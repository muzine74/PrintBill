--  SELECT * FROM dbo.fn_GetEmployeesWithCompanies() ORDER BY EmployeeId;

CREATE FUNCTION [dbo].[fn_GetEmployeesWithCompanies]()
RETURNS TABLE
AS
RETURN
(
    SELECT
        e.Name,
        e.EmployeeId,
        c.*
    FROM dbo.Employees e
    INNER JOIN dbo.EmployeeCompanies ec
        ON e.EmployeeId = ec.EmployeeId
    INNER JOIN dbo.Companies c
        ON ec.CompanyId = c.CompanyId
);
GO

CREATE FUNCTION [dbo].[fn_GetCompaniesByEmployee]
(
    @EmployeeId UNIQUEIDENTIFIER
)
RETURNS TABLE
AS
RETURN
(
    SELECT
        e.Name,
        e.EmployeeId,
        c.*
    FROM dbo.Employees e
    INNER JOIN dbo.EmployeeCompanies ec
        ON e.EmployeeId = ec.EmployeeId
    INNER JOIN dbo.Companies c
        ON ec.CompanyId = c.CompanyId
    WHERE e.EmployeeId = @EmployeeId
);
GO
